\chapter{Overview}
This codebook documents the \texttt{hetmacro} package for numerical tools in macroeconomics with heterogeneity. The package is designed to be self-contained and modular, with a foundations-first architecture.

\chapter{Foundational Modules}

\section{\texttt{grids.py}}
\begin{itemize}
  \item \texttt{GridSpec}: specification for one grid dimension
  \item \texttt{make\_grid\_1d}: flexible spacing and concentration
  \item \texttt{make\_grid\_nd}: build multi-dimensional grids
  \item \texttt{cartesian\_product}, \texttt{gridmake}
  \item \texttt{make\_asset\_grid}, \texttt{double\_exponential\_grid}, \texttt{log\_spaced\_grid}
\end{itemize}

\section{\texttt{quadrature.py}}
Gaussian and Newton--Cotes quadrature rules.
\begin{itemize}
  \item \texttt{qnwlege}, \texttt{qnwcheb}, \texttt{qnwnorm}
  \item \texttt{qnwlogn}, \texttt{qnwunif}, \texttt{qnwbeta}, \texttt{qnwgamma}
  \item \texttt{qnwsimp}, \texttt{qnwtrap}
  \item \texttt{qnwnorm\_mv}
\end{itemize}

\section{\texttt{interpolation.py}}
\begin{itemize}
  \item \texttt{interp\_linear}, \texttt{interp\_bilinear}
  \item \texttt{get\_lottery}
  \item Spline and Chebyshev tools: \texttt{spline\_coef}, \texttt{cheb\_coef}, \texttt{cheb\_eval}
\end{itemize}

\section{\texttt{optimize.py}}
Root-finding and optimization wrappers.
\begin{itemize}
  \item \texttt{bisect}, \texttt{brentq}, \texttt{newton}, \texttt{secant}, \texttt{broyden}
  \item \texttt{golden\_search}, \texttt{brent\_min}, \texttt{nelder\_mead}
\end{itemize}

\section{\texttt{markov.py}}
\begin{itemize}
  \item \texttt{rouwenhorst}, \texttt{tauchen}
  \item \texttt{stationary\_distribution}, \texttt{simulate\_markov}
\end{itemize}

\section{\texttt{utils.py}}
\begin{itemize}
  \item \texttt{crra\_utility}, \texttt{crra\_marginal}, \texttt{crra\_inverse\_marginal}
  \item \texttt{ces\_aggregator}
  \item \texttt{compute\_fixed\_point}, \texttt{numerical\_derivative}, \texttt{numerical\_jacobian}
  \item \texttt{tic}, \texttt{toc}
\end{itemize}
